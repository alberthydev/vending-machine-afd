Implementação por if/switch

A cada moeda inserida, soma-se o valor ao estado atual e decide-se na hora o próximo estado: chegou exatamente a 30 → libera o produto; passou de 30 → trava. Direto e fácil de ler para um autômato deste tamanho.

function simularPorSwitch(moedas) {
  let estado = 0;
  for (const moeda of moedas) {
    if (estado === 30 || estado === "30+") continue; // self-loop
    switch (moeda) {
      case 5: case 10: case 25: {
        const novoValor = estado + moeda;
        if (novoValor === 30) estado = 30;
        else if (novoValor > 30) estado = "30+";
        else estado = novoValor;
        break;
      }
      default:
        throw new Error("Moeda inválida: " + moeda);
    }
  }
  return estado; // 30 = produto liberado, "30+" = travado
}
